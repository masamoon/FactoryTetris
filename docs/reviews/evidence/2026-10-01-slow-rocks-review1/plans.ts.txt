import { SLOTS, ORES } from '../src/rockhopper/config';
import { useSector } from '../src/rockhopper/sector';
import { generateRock } from '../src/rockhopper/sim';
const DEPTH: Record<number, number> = { 1: 60, 2: 80, 3: 100, 4: 120 };
const acc: Record<string, { flat: number[]; lay: number[]; oreShare: number[] }> = {};
for (let seed = 1; seed <= 400; seed++) {
  useSector(seed, true);
  for (let i = 0; i < 8; i++) {
    const def = SLOTS[i]; const rock = generateRock(i, 0, seed); const D = DEPTH[def.tier];
    let flat = 0, lay = 0, oreLay = 0, allLay = 0;
    for (let k = 0; k < rock.cells.length; k++) { const c = rock.cells[k]; if (!c) continue;
      const gx = (k % rock.w) - (rock.r + 1), gy = Math.floor(k / rock.w) - (rock.r + 1); const d = Math.hypot(gx, gy), rho = Math.min(1, d / rock.r);
      const L = d > rock.r - 1.5 ? 1 : 1 + Math.round((D - 1) * Math.max(0, (1 - rho) / (1 - 1.5 / rock.r)));
      flat += (ORES as any)[c].value; lay += (ORES as any)[c].value * L; allLay += L; if (c > 1) oreLay += L; }
    const key = `T${def.tier} ${def.veins!.kind}`;
    (acc[key] ??= { flat: [], lay: [], oreShare: [] }); acc[key].flat.push(flat); acc[key].lay.push(lay); acc[key].oreShare.push(oreLay / allLay);
  }
}
const mean = (a: number[]) => a.reduce((x, y) => x + y, 0) / a.length;
const tierMean: Record<string, number> = {}; const tierFlat: Record<string, number> = {};
for (const t of ['T1', 'T2', 'T3', 'T4']) { const ks = Object.keys(acc).filter((k) => k.startsWith(t)); tierMean[t] = mean(ks.flatMap((k) => acc[k].lay)); tierFlat[t] = mean(ks.flatMap((k) => acc[k].flat)); }
for (const k of Object.keys(acc).sort()) { const t = k.slice(0, 2); const a = acc[k];
  console.log(`${k.padEnd(12)} n=${String(a.lay.length).padStart(3)} flat value ${(100 * mean(a.flat) / tierFlat[t]).toFixed(0)}% of tier mean, layered ${(100 * mean(a.lay) / tierMean[t]).toFixed(0)}%, ore share of layers ${(100 * mean(a.oreShare)).toFixed(0)}%`); }
