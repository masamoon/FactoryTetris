import { runBot } from './bot';
import { ORES } from './src/rockhopper/config';
for (const seed of [1,2,3,4,5,6,7,8]) {
  let n = 0, ore = 0, val = 0;
  runBot({ minutes: 20, laser: true, seed, sector: true, onTick: (s) => {
    for (const e of s.events) if (e.type === 'break' && e.by === 'drill') { n++; if (e.ore !== 1) ore++; val += ORES[e.ore].value; }
  } });
  console.log(`s${seed}S drill chunks ${n} ore share ${(100*ore/n).toFixed(0)}% value/chunk ${(val/n).toFixed(2)}`);
}
